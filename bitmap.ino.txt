void setup() {
  matrix.setBrightness(50);
  matrix.clear();
  for (int i = 0; i < 1000; i++) {
    int delta = i % (spritepain_w*2)-spritepain_w;
    matrix.drawBitmap8(delta, delta, spritepain, spritepain_w, spritepain_h);
    matrix.show();
    delay(500);
    matrix.clear();
  }
